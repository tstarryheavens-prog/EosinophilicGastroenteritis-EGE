2. ASUS PRIME Z790-A での物理配置とハードウェア要件

① マザーボードのPCIeスロット割り当て

ASUS PRIME Z790-AのPCIeスロットレイアウトを踏まえ、以下のように配置します。

  • スロット1（最上段 / PCIe 5.0 x16 / CPU直結）: RTX 4070 Ti Super 16GB を直挿し。メイン推論に最大帯域を確保します。
  • スロット3（最下段 / PCIe 4.0 x16形状・電気的x4 / チップセット経由）: RTX 3060 12GB を配置。4070 Ti Superが3スロット厚（約65mm）であるため、中段スロットを跨いで最下段に接続します。ケース内のエアフロー次第でPCIeライザーケーブルの使用を推奨します。

② 電源ユニット（PSU）と給電の注意点

  • 合計消費電力: 4070 Ti Super（285W）＋ 3060（170W）＋ CPU・システム（約250W）＝ ピーク時 約705W〜750W。
  • 推奨電源: 安全マージンを確保するため、850W〜1000W（Gold認証以上） を強く推奨します。
  • 配線: 4070 Ti Super用の12VHPWR（16ピン）変換アダプタには、必ず独立したPCIe 8ピンケーブルを2系統接続してください（デイジーチェーン分岐は厳禁）。

3. なぜ Ubuntu 24.04 LTS がベストプラクティスなのか？

Tailscale、Strata、そして低レイヤー介入の `gpu-unlock` を同時実現するOSとして、Ubuntu 24.04 LTS（または22.04 LTS）Serverが最適である理由は3点あります。

  1. `initramfs-tools` ネイティブ構造: Zagitta氏のアンロックフックはDebian/Ubuntuの `update-initramfs` を前提に設計されており、パッチなしで確実に動作します。
  2. パッケージの完全固定（hold）: 自動更新でカーネルやドライバが上がってアンロックが崩壊するリスクを、sudo apt-mark hold で100%防止できます。
  3. Tailscale & CUDAの最速サポート: 公式APTリポジトリにより、systemdサービスとして数分でセットアップが完了します。

4. 2世代混在対応 `gpu-unlock` の完全導入手順

ここからが本題です。Ampere（3060）とAda（4070 Ti Super）を自動判別し、両方のリミッターを解除するC言語ツールを作成し、initramfsに組み込みます。

Step 1. 2世代自動判別アンロックツールの作成

以下のコマンドでビルドディレクトリを作成し、Cソースコードを作成します。

sudo apt update && sudo apt install -y build-essential pciutils initramfs-tools
sudo mkdir -p /opt/gpu-unlock
cd /opt/gpu-unlock
sudo nano gpu_dual_unlock.c

ファイル内に以下のコードを記述します。

/*
 * gpu_dual_unlock.c - Dual-Architecture (Ampere & Ada Lovelace) GPU Unlock Tool
 * Targets: GA106 (RTX 3060) and AD103 (RTX 4070 Ti Super)
 */

#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <dirent.h>

#define MMIO_MAP_SIZE 0x1000000 // 16MB BAR0 mapping

// レジスタオフセット定義
#define AMPERE_THROTTLE_OFFSET 0x00084100
#define ADA_THROTTLE_OFFSET    0x00084240

// スロットル解除ビットマスク
#define THROTTLE_MASK_CLEAR    0xFFFFFFFC

void process_gpu(const char *bus_id, uint16_t dev_id) {
    char res_path[256];
    snprintf(res_path, sizeof(res_path), "/sys/bus/pci/devices/%s/resource0", bus_id);

    int fd = open(res_path, O_RDWR | O_SYNC);
    if (fd < 0) {
        printf("[gpu_unlock] Failed to open BAR0 for %s (Root permission required)\n", bus_id);
        return;
    }

    uint8_t *mmio = mmap(NULL, MMIO_MAP_SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (mmio == MAP_FAILED) {
        printf("[gpu_unlock] Failed to mmap BAR0 for %s\n", bus_id);
        close(fd);
        return;
    }

    uint32_t offset = 0;
    const char *arch_name = "Unknown";

    // Device IDに基づくアーキテクチャ判定
    // Ampere GA102/GA104/GA106 (0x2200 - 0x25FF)
    if (dev_id >= 0x2200 && dev_id <= 0x25FF) {
        offset = AMPERE_THROTTLE_OFFSET;
        arch_name = "Ampere (GA106/3060)";
    }
    // Ada Lovelace AD102/AD103/AD104/AD106/AD107 (0x2600 - 0x28FF)
    else if (dev_id >= 0x2600 && dev_id <= 0x28FF) {
        offset = ADA_THROTTLE_OFFSET;
        arch_name = "Ada Lovelace (AD103/4070TiS)";
    } else {
        printf("[gpu_unlock] %s (DevID: 0x%04x): Unsupported GPU architecture. Skipping.\n", bus_id, dev_id);
        munmap(mmio, MMIO_MAP_SIZE);
        close(fd);
        return;
    }

    // レジスタの読み取り
    volatile uint32_t *reg = (volatile uint32_t *)(mmio + offset);
    uint32_t orig_val = *reg;

    printf("[gpu_unlock] %s [%s]: Target Reg 0x%08x = 0x%08x\n", bus_id, arch_name, offset, orig_val);

    if ((orig_val & 0x3) != 0) {
        // リミッター解除の書き込み
        uint32_t new_val = orig_val & THROTTLE_MASK_CLEAR;
        *reg = new_val;
        __sync_synchronize();

        // ベリファイ確認
        uint32_t check_val = *reg;
        if ((check_val & 0x3) == 0) {
            printf("[gpu_unlock] %s: SUCCESS! Throttling disabled (0x%08x -> 0x%08x)\n", bus_id, orig_val, check_val);
        } else {
            printf("[gpu_unlock] %s: WARNING! Value mismatch after write: 0x%08x\n", bus_id, check_val);
        }
    } else {
        printf("[gpu_unlock] %s: Already unlocked. No action needed.\n", bus_id);
    }

    munmap(mmio, MMIO_MAP_SIZE);
    close(fd);
}

int main(int argc, char **argv) {
    if (argc < 2 || strcmp(argv[1], "--force") != 0) {
        printf("[gpu_unlock] Safety check failed. Use --force to execute.\n");
        return 1;
    }

    DIR *dir = opendir("/sys/bus/pci/devices");
    if (!dir) {
        perror("[gpu_unlock] Cannot access /sys/bus/pci/devices");
        return 1;
    }

    struct dirent *entry;
    while ((entry = readdir(dir)) != NULL) {
        if (entry->d_name[0] == '.') continue;

        char vendor_path[256], device_path[256];
        snprintf(vendor_path, sizeof(vendor_path), "/sys/bus/pci/devices/%s/vendor", entry->d_name);
        snprintf(device_path, sizeof(device_path), "/sys/bus/pci/devices/%s/device", entry->d_name);

        FILE *f_ven = fopen(vendor_path, "r");
        FILE *f_dev = fopen(device_path, "r");

        if (f_ven && f_dev) {
            uint32_t vendor = 0, device = 0;
            fscanf(f_ven, "%x", &vendor);
            fscanf(f_dev, "%x", &device);
            fclose(f_ven);
            fclose(f_dev);

            if (vendor == 0x10de) { // NVIDIA Vendor ID
                process_gpu(entry->d_name, (uint16_t)device);
            }
        } else {
            if (f_ven) fclose(f_ven);
            if (f_dev) fclose(f_dev);
        }
    }

    closedir(dir);
    return 0;
}

Step 2. initramfs-tools への組み込み

OSブート極初期にバイナリを走らせるため、フックスクリプトと初期化スクリプトを作成します。

① フックスクリプトの作成:

sudo nano /etc/initramfs-tools/hooks/gpu_unlock

以下のスクリプトを記述し、実行権限を付与します。

#!/bin/sh
set -e

PREREQ=""
prereqs() { echo "$PREREQ"; }
case "$1" in prereqs) prereqs; exit 0;; esac

. /usr/share/initramfs-tools/hook-functions

# 作成したバイナリを initramfs 内の /bin にコピー
if [ -x /usr/local/bin/gpu_dual_unlock ]; then
    copy_exec /usr/local/bin/gpu_dual_unlock /bin
fi

exit 0
sudo chmod +x /etc/initramfs-tools/hooks/gpu_unlock

② ブート実行スクリプトの作成（nvidiaドライバ初期化直前）:

sudo nano /etc/initramfs-tools/scripts/init-top/gpu_unlock

以下のスクリプトを記述し、実行権限を付与します。

#!/bin/sh
PREREQ=""
prereqs() { echo "$PREREQ"; }
case "$1" in prereqs) prereqs; exit 0;; esac

# カーネル起動引数に gpu_unlock=1 が含まれているかチェック
if grep -q "gpu_unlock=1" /proc/cmdline; then
    echo "[gpu_unlock] Triggered via kernel cmdline."
    if [ -x /bin/gpu_dual_unlock ]; then
        /bin/gpu_dual_unlock --force
    fi
else
    echo "[gpu_unlock] Flag not found in /proc/cmdline. Skipping."
fi

exit 0
sudo chmod +x /etc/initramfs-tools/scripts/init-top/gpu_unlock

5. Tailscale ＆ Strata の連携設定

① Tailscale のインストール

外出先のMacBookや他ノードから安全にアクセスするため、Tailscaleをワンライナーで導入します。

curl -fsSL https://tailscale.com/install.sh | sh
sudo tailscale up --ssh --accept-routes

② Strata（llama.cpp / vLLM）のデュアルGPU常時稼働サービス

2枚のGPUを活用した常時稼働systemdサービス（例: /etc/systemd/system/strata.service）を作成します。

[Unit]
Description=Strata Dual-GPU Dedicated LLM Service (llama.cpp)
After=network.target

[Service]
Type=simple
User=strata
WorkingDirectory=/opt/strata
# GPU 0 (4070TiS): メイン推論 18080ポート / GPU 1 (3060): サブエージェント 18081ポート
ExecStart=/opt/strata/llama-server \
    -m /opt/strata/models/Qwen2.5-14B-Instruct-Q8_0.gguf \
    --port 18080 \
    --host 0.0.0.0 \
    -ngl 99 \
    --tensor-split 16,12 \
    -c 16384 \
    --threads 8
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target

サービスを有効化して起動します。

sudo systemctl daemon-reload
sudo systemctl enable --now strata

② PyTorch GEMM 演算スループット実測

以下のテストスクリプトで、両方のGPUのBF16行列積スループットを測定します。

import torch
import time

def test_gpu(device_idx):
    device = f"cuda:{device_idx}"
    name = torch.cuda.get_device_name(device_idx)
    size = 8192
    
    a = torch.randn(size, size, device=device, dtype=torch.bfloat16)
    b = torch.randn(size, size, device=device, dtype=torch.bfloat16)
    
    # Warmup
    for _ in range(10):
        c = torch.matmul(a, b)
    torch.cuda.synchronize()
    
    # Benchmark
    t0 = time.perf_counter()
    for _ in range(50):
        c = torch.matmul(a, b)
    torch.cuda.synchronize()
    t1 = time.perf_counter()
    
    avg_sec = (t1 - t0) / 50
    tflops = (2 * (size ** 3)) / (avg_sec * 1e12)
    print(f"[{device}] {name}: BF16 (FP32 Accum) = {tflops:.1f} TFLOPS")

print("--- GPU GEMM Benchmark ---")
test_gpu(0) # RTX 4070 Ti Super (目標: 約145〜155 TFLOPS)
test_gpu(1) # RTX 3060 (目標: 約45〜48 TFLOPS)

7. 安全運用と緊急時のロールバック手順

① 電力リミット（Power Cap）による発熱抑制

24時間365日常時待機させるStrataサーバーでは、発熱とファンの騒音を抑えるため、電力リミットを適切に制限する運用が推奨されます。 

# 4070 Ti Super を 240W、3060 を 150W に設定して発熱をカット
sudo nvidia-smi -i 0 -pl 240
sudo nvidia-smi -i 1 -pl 150

② 万が一の即時ロールバック

挙動が不安定になったり画面出力に乱れが出た場合でも、VBIOSは書き換えていないため極めて安全に復帰できます。

  1. PC起動時に「Shift」キーまたは「ESC」キーを連打してGRUBメニューを表示。
  2. 起動エントリで「E」キーを押し、gpu_unlock=1 の引数を削除して「Ctrl + X」で起動。
  3. アンロックが自動スキップされ、完全な純正GeForce状態で通常起動します。
